cmake_minimum_required(VERSION 3.16)
project(payload_processing)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(payload_interfaces REQUIRED)
find_package(JPEG REQUIRED)
add_library(process src/process.cpp)
target_compile_features(process PUBLIC cxx_std_17)
target_compile_options(process PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(process PUBLIC include)
ament_target_dependencies(process sensor_msgs)
target_link_libraries(process JPEG::JPEG)
add_executable(processing_node src/processing_node.cpp)
target_compile_features(processing_node PUBLIC cxx_std_17)
target_compile_options(processing_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(processing_node rclcpp payload_interfaces)
target_link_libraries(processing_node process)
install(TARGETS process processing_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib
  RUNTIME DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_process test/test_process.cpp)
  target_link_libraries(test_process process)
endif()
ament_package()
