cmake_minimum_required(VERSION 3.16)
project(payload_smoke)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(smoke src/smoke.cpp)
target_compile_features(smoke PUBLIC cxx_std_17)
target_compile_options(smoke PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(smoke rclcpp std_msgs)
install(TARGETS smoke DESTINATION lib/${PROJECT_NAME})
ament_package()
