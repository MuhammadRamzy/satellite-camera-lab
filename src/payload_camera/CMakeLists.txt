cmake_minimum_required(VERSION 3.16)
project(payload_camera)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(payload_interfaces REQUIRED)
add_executable(camera_node src/camera_node.cpp)
target_compile_features(camera_node PUBLIC cxx_std_17)
target_compile_options(camera_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(camera_node PRIVATE include)
ament_target_dependencies(camera_node rclcpp rclcpp_lifecycle rclcpp_action lifecycle_msgs sensor_msgs payload_interfaces)
install(TARGETS camera_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES ../../config/pipeline.yaml DESTINATION share/${PROJECT_NAME}/config)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_modes test/test_modes.cpp)
  target_include_directories(test_modes PRIVATE include)
  target_compile_features(test_modes PUBLIC cxx_std_17)
endif()
ament_package()
